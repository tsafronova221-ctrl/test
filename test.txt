#!/bin/bash

# Скрипт для подсчета четных и нечетных чисел
# Использование: ./script.sh число1 число2 число3 ...

# Проверка на наличие аргументов
if [ $# -eq 0 ]; then
    echo "Ошибка: Не переданы числа для анализа"
    echo "Использование: $0 число1 число2 число3 ..."
    exit 1
fi

# Инициализация счетчиков
even_count=0  # счетчик четных чисел
odd_count=0   # счетчик нечетных чисел

echo "Анализ чисел: $@"
echo "----------------"

# Обработка каждого аргумента
for number in "$@"; do
    # Проверка, является ли аргумент числом
    if ! [[ "$number" =~ ^-?[0-9]+$ ]]; then
        echo "Пропускаем '$number' - это не целое число"
        continue
    fi
    
    # Проверка четности
    if [ $((number % 2)) -eq 0 ]; then
        echo "  $number - четное"
        ((even_count++))
    else
        echo "  $number - нечетное"
        ((odd_count++))
    fi
done

echo "----------------"
echo "Итог:"
echo "  Четных чисел: $even_count"
echo "  Нечетных чисел: $odd_count"
echo ""

# Определение результата
if [ $even_count -gt $odd_count ]; then
    echo "Результат: Четных больше"
elif [ $odd_count -gt $even_count ]; then
    echo "Результат: Нечетных больше"
else
    echo "Результат: Четных и нечетных поровну ($even_count каждого)"
fi
