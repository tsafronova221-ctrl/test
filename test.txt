#!/bin/bash
# Скрипт обработки чисел с исключением 3-й и 6-й переменных

echo "========================================"
echo "    ОБРАБОТКА ЧИСЕЛ"
echo "========================================"
echo "Введите числа через пробел (не менее 6 для демонстрации):"
echo "Пример: 1 2 3 4 5 6 7 8"
echo ""

# Считываем ввод пользователя
read -p "Числа: " input

# Проверка на пустой ввод
if [ -z "$input" ]; then
    echo "Ошибка: Не введены числа"
    exit 1
fi

# Разбиваем ввод на массив
numbers=($input)
total=${#numbers[@]}

echo ""
echo "Получено чисел: $total"
echo ""

# Инициализация переменных
declare -a variables  # Массив для хранения переменных
product=1             # Произведение четных (инициализируем 1)
sum=0                 # Сумма нечетных

# Создаем переменные и выполняем расчеты
for ((i=0; i<total; i++)); do
    index=$((i+1))
    number="${numbers[i]}"
    
    # Проверка, что введено число
    if ! [[ "$number" =~ ^-?[0-9]+$ ]]; then
        echo "Пропускаем '$number' (не число) на позиции $index"
        continue
    fi
    
    # Создаем переменную с именем var_номер
    var_name="var_$index"
    declare "$var_name=$number"
    variables+=("$var_name")
    
    echo "Переменная $var_name = $number"
    
    # Пропускаем 3-ю и 6-ю переменные в расчетах
    if [ $index -eq 3 ] || [ $index -eq 6 ]; then
        echo "  ⚠️  Пропуск в расчетах (3-я или 6-я переменная)"
        continue
    fi
    
    # Проверяем четность и выполняем соответствующие операции
    if [ $((number % 2)) -eq 0 ]; then
        echo "  ✓ Четное → идет в произведение"
        product=$((product * number))
    else
        echo "  ✓ Нечетное → идет в сумму"
        sum=$((sum + number))
    fi
done

echo ""
echo "========================================"
echo "        РЕЗУЛЬТАТЫ РАСЧЕТОВ"
echo "========================================"

# Выводим все созданные переменные
echo "Созданные переменные:"
for var in "${variables[@]}"; do
    eval "value=\$$var"
    echo "  $var = $value"
done

echo ""
echo "Результаты расчетов (без учета 3-й и 6-й переменных):"
echo "  Произведение четных чисел: $product"
echo "  Сумма нечетных чисел: $sum"
echo ""

# Дополнительная информация
echo "Статистика:"
echo "  Всего переменных: ${#variables[@]}"
echo "  Учтено в расчетах: $(( ${#variables[@]} - (total >= 3 ? 1 : 0) - (total >= 6 ? 1 : 0) ))"
echo "  Пропущено (3-я и 6-я): $(( (total >= 3 ? 1 : 0) + (total >= 6 ? 1 : 0) ))"
