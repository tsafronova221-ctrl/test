#!/bin/bash
# Скрипт для вывода системной информации с проверкой времени

echo "========================================"
echo "       СИСТЕМНАЯ ИНФОРМАЦИЯ"
echo "========================================"

# Получаем текущую дату и время
current_date=$(date '+%d.%m.%Y')
current_time=$(date '+%H:%M:%S')
hour=$(date '+%H')  # Только часы (24-часовой формат)

echo "Дата: $current_date"
echo "Время: $current_time"
echo ""

# Количество пользователей в системе
users_count=$(who | wc -l)
echo "Количество пользователей в системе: $users_count"
echo "Список пользователей:"
who | awk '{print "  " $1 " (" $2 ") - " $3 " " $4}'
echo ""

# Проверка условий по времени
echo "========================================"
echo "         ПРОВЕРКА ВРЕМЕНИ"
echo "========================================"

# Преобразуем часы в число для сравнения
hour_num=$((10#$hour))  # 10# чтобы избежать ошибок с ведущими нулями

if [ $hour_num -ge 3 ] && [ $hour_num -ge 12 ]; then
    echo "⚠️  УВЕДОМЛЕНИЕ: Время больше 3-х часов И больше 12:00"
    echo "   Сейчас $current_time"
    
    if [ $hour_num -ge 12 ] && [ $hour_num -lt 18 ]; then
        echo "   Это день (после полудня)"
    elif [ $hour_num -ge 18 ]; then
        echo "   Это вечер"
    fi
else
    echo "✓ Условие не выполняется:"
    echo "  - Время должно быть > 3 часов"
    echo "  - И время должно быть > 12:00"
    
    if [ $hour_num -lt 3 ]; then
        echo "  Сейчас ночь/раннее утро (до 3:00)"
    elif [ $hour_num -lt 12 ]; then
        echo "  Сейчас утро (до 12:00)"
    fi
fi
