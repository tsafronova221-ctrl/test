#!/bin/bash
# file-finder.sh - компактная версия

# Проверка аргументов
if [ $# -ne 2 ]; then
    echo "Использование: $0 <директория> <тип_файла>"
    echo "Пример: $0 /home/user ascii"
    exit 1
fi

search_dir="$1"
file_type="$2"
output_file="${file_type}.txt"

# Проверка директории
if [ ! -d "$search_dir" ] || [ ! -r "$search_dir" ]; then
    echo "Ошибка: директория '$search_dir' недоступна"
    exit 1
fi

# Поиск файлов
echo "Поиск файлов типа '$file_type' в '$search_dir'..."
count=0

# Используем find для рекурсивного поиска
find "$search_dir" -type f -print0 2>/dev/null | while IFS= read -r -d $'\0' file; do
    if file_info=$(file -b --mime-type "$file" 2>/dev/null); then
        if echo "$file_info" | grep -qi "$file_type"; then
            echo "$file"
            ((count++))
        fi
    fi
done > "$output_file"

# Подсчет результатов
found_count=$(wc -l < "$output_file" 2>/dev/null || echo 0)

if [ $found_count -gt 0 ]; then
    echo "Найдено файлов: $found_count"
    echo "Результаты сохранены в: $output_file"
else
    echo "Файлов не найдено"
    > "$output_file"
fi
